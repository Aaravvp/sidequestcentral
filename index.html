    questsRef.onSnapshot((snapshot) => {

        quests = [];

        snapshot.forEach((doc) => {

            const data = doc.data();

            quests.push({

                id: doc.id,

                ...data,

                joinedByMe: Array.isArray(data.joinedUsers) && data.joinedUsers.includes(myUserId)

            });

        });

        renderFeed();

    });


    function formatTime12hr(timeStr) {

        const [h, m] = timeStr.split(':').map(Number);

        const ampm = h >= 12 ? 'pm' : 'am';

        const hour12 = h % 12 || 12;

        const minStr = m < 10 ? `0${m}` : m;

        return `${hour12}:${minStr} ${ampm}`;

    }


    function getQuestTimestamp(q) {

        if (q.isoDateTime) {

            const dateObj = new Date(q.isoDateTime);

            if (!isNaN(dateObj.getTime())) return dateObj;

        }

        return new Date(2099, 11, 31);

    }


    function checkScheduleClash(targetTimestamp) {

        const targetTime = targetTimestamp.getTime();

        const duration = 90 * 60 * 1000;

        for (let q of quests) {

            if (q.joinedByMe) {

                const qTime = getQuestTimestamp(q).getTime();

                if (Math.abs(targetTime - qTime) < duration) return q;

            }

        }

        return null;

    }


    function setDefaultFormDate() {

        const now = new Date();

        const year = now.getFullYear();

        const month = String(now.getMonth() + 1).padStart(2, '0');

        const day = String(now.getDate()).padStart(2, '0');

        document.getElementById('questDate').value = `${year}-${month}-${day}`;

        document.getElementById('questDate').min = `${year}-${month}-${day}`;


        let hours = now.getHours() + 1;

        if (hours >= 24) hours = 23;

        const hoursStr = String(hours).padStart(2, '0');

        document.getElementById('questTime').value = `${hoursStr}:00`;

    }


    function renderFeed() {

        const feed = document.getElementById('questFeed');

        feed.innerHTML = '';

        const now = new Date();


        let filtered = quests.filter(q => {

            const qDate = getQuestTimestamp(q);

            if (qDate < new Date(now.getTime() - (2 * 60 * 60 * 1000))) return false;

            if (activeCat !== 'all' && q.cat !== activeCat) return false;


            const isSameDay = (d1, d2) => d1.getFullYear() === d2.getFullYear() && d1.getMonth() === d2.getMonth() && d1.getDate() === d2.getDate();

            const tomorrow = new Date(now);

            tomorrow.setDate(now.getDate() + 1);


            if (activeDay === 'today') return isSameDay(qDate, now);

            if (activeDay === 'tomorrow') return isSameDay(qDate, tomorrow);

            if (activeDay === 'weekend') {

                const day = qDate.getDay();

                return day === 0 || day === 6;

            }


            return true;

        });


        filtered.sort((a, b) => getQuestTimestamp(a) - getQuestTimestamp(b));


        if (filtered.length === 0) {

            feed.innerHTML = `<div style="text-align:center; font-weight:900; padding:40px 10px; font-size:1.2rem;">no sidequests live yet.<br><span style="font-size:0.9rem; font-weight:700;">be the first to drop one above.</span></div>`;

            return;

        }


        filtered.forEach(q => {

            feed.innerHTML += `

                <div class="card" onclick="openModal('${q.id}')">

                    <div class="card-header">

                        <div class="card-title">${q.title}</div>

                        <span class="card-tag">${q.cat}</span>

                    </div>

                    <div class="card-meta">

                        <span class="card-time-badge">🕒 ${q.time}</span>

                        <span>📍 ${q.loc.split(',')[0]}</span>

                        <span>👤 ${q.host}</span>

                    </div>

                    <div class="card-footer">

                        <span class="slots">${q.joinedCount || 1} down</span>

                        <button class="btn-join ${q.joinedByMe ? 'joined' : ''}" 

                                onclick="event.stopPropagation(); toggleJoinQuest('${q.id}')">

                            ${q.joinedByMe ? 'you\'re down ✓' : 'i\'m down'}

                        </button>

                    </div>

                </div>

            `;

        });

    }

    async function createQuest() {

        if (!requireOnboarding()) return;


        const hostInput = document.getElementById('questHost').value.trim();

        const title = document.getElementById('questTitle').value.trim();

        const loc = document.getElementById('questLoc').value.trim();

        const dateInput = document.getElementById('questDate').value;

        const timeInput = document.getElementById('questTime').value;

        const deets = document.getElementById('questDeets').value.trim();

        const cat = document.getElementById('questCat').value;


        if (!isValidTextContent(hostInput, 2, 40)) return alert('please enter a valid host name!');

        if (!isValidTextContent(title, 3, 60)) return alert('please enter a real quest title!');

        if (!isValidTextContent(loc, 3, 80)) return alert('please enter a valid location!');

        if (!dateInput || !timeInput) return alert('please select a valid date and time!');


        const targetDate = new Date(`${dateInput}T${timeInput}`);

        const now = new Date();


        if (isNaN(targetDate.getTime())) return alert('invalid date or time selected!');

        if (targetDate <= now) return alert('you cannot post a sidequest in the past!');


        const isSameDay = (d1, d2) => d1.getFullYear() === d2.getFullYear() && d1.getMonth() === d2.getMonth() && d1.getDate() === d2.getDate();

        const tomorrow = new Date(now);

        tomorrow.setDate(now.getDate() + 1);


        let dayPrefix = '';

        if (isSameDay(targetDate, now)) {

            dayPrefix = 'today @ ';

        } else if (isSameDay(targetDate, tomorrow)) {

            dayPrefix = 'tomorrow @ ';

        } else {

            const options = { month: 'short', day: 'numeric' };

            dayPrefix = `${targetDate.toLocaleDateString('en-US', options).toLowerCase()} @ `;

        }


        const displayTime = `${dayPrefix}${formatTime12hr(timeInput)}`;


        const clashingQuest = checkScheduleClash(targetDate);

        if (clashingQuest) {

            alert(`clash alert! this overlaps with "${clashingQuest.title}" (${clashingQuest.time}) which you're already attending.`);

            return;

        }


        const newQuest = {

            title: title.toLowerCase(),

            loc: loc,

            time: displayTime,

            isoDateTime: targetDate.toISOString(),

            deets: deets.toLowerCase(),

            cat: cat,

            joinedCount: 1,

            joinedUsers: [myUserId],

            host: hostInput.toLowerCase(),

            userEmail: localStorage.getItem('sq_user_email'),

            campus: localStorage.getItem('sq_user_campus') || '',

            createdAt: firebase.firestore.FieldValue.serverTimestamp()

        };


        await questsRef.add(newQuest);

        toggleForm();

        document.getElementById('questTitle').value = '';

        document.getElementById('questLoc').value = '';

        document.getElementById('questDeets').value = '';

    }


    async function executeToggleJoin(id) {

        const quest = quests.find(q => q.id === id);

        if (!quest) return;

        const docRef = questsRef.doc(id);


        if (quest.joinedByMe) {

            await docRef.update({

                joinedUsers: firebase.firestore.FieldValue.arrayRemove(myUserId),

                joinedCount: firebase.firestore.FieldValue.increment(-1)

            });

        } else {

            const questDate = getQuestTimestamp(quest);

            const clashingQuest = checkScheduleClash(questDate);

            if (clashingQuest) {

                alert(`clash alert! you're already down for "${clashingQuest.title}" around ${clashingQuest.time}. leave that one first!`);

                return;

            }

            await docRef.update({

                joinedUsers: firebase.firestore.FieldValue.arrayUnion(myUserId),

                joinedCount: firebase.firestore.FieldValue.increment(1)

            });

        }

    }


    async function toggleJoinQuest(id) {

        requireOnboarding(() => executeToggleJoin(id));

    }
